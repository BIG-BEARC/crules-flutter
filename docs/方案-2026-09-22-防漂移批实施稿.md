## 1 新建 `hooks/_common.py`（全文）

```python
#!/usr/bin/env python3
"""hooks 共享实现（1.0.39 · 裁决单-2026-09-22 D-1）

三件各 hook 逐字重复的事收到一处：找项目根 / sid 净化 / stdout·stderr·stdin 钉 UTF-8。
此前各文件内联拷贝（pending/stop/compliance 三份 find_project_root+MAX_SID，
stop/deny/compliance 三份 _hook_utf8_streams）——漂移只靠注释约束，无机械保障。
分发说明：hook 以 `python3 <路径>/hooks/x.py` 整目录随 plugin 安装，脚本同目录自动在
sys.path[0]，`import _common` 直接可用，无路径样板。
"""
import os
import re
import sys

MAX_SID = 80  # sid 作文件名时的截断长度


def find_project_root(start=None):
    """自 start（默认 cwd）向上找 .claude/memory/NAVIGATION.md，找到其上一级即项目根"""
    d = os.path.abspath(start or os.getcwd())
    while True:
        if os.path.isfile(os.path.join(d, ".claude", "memory", "NAVIGATION.md")):
            return d
        parent = os.path.dirname(d)
        if parent == d:
            return None
        d = parent


def sanitize_sid(raw):
    """session_id 净化为文件名片段：非 [A-Za-z0-9_-] 换 '-'，截断 MAX_SID"""
    return re.sub(r"[^A-Za-z0-9_-]", "-", str(raw or ""))[:MAX_SID]


def _hook_utf8_streams():
    """非 UTF-8 码页宿主下 print/读崩或失真——三流显式钉 UTF-8（1.0.21 P0-A 实证）"""
    for name in ("stdin", "stdout", "stderr"):
        s = getattr(sys, name, None)
        if s is None:
            continue
        try:
            s.reconfigure(encoding="utf-8", errors="replace")
            continue
        except Exception:
            pass
        try:   # Python < 3.7 无 reconfigure：退到重包 TextIOWrapper
            import io
            setattr(sys, name, io.TextIOWrapper(s.buffer, encoding="utf-8", errors="replace"))
        except Exception:
            pass
```
